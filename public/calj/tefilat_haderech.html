תְפִלַּת הַדֶּרֶךְ

    יְהִי רָצוֹן מִלְפָנֶיךָ יי אֱלֹהֵינוּ וֵאלֹהֵי אֲבוֹתֵינוּ, שֶׁתּוֹלִיכֵנוּ לְשָׁלוֹם וְתַצְעִידֵנוּ לְשָׁלוֹם (וְתִסְמְכֵנוּ לְשָׁלוֹם) וְתַדְרִיכֵנוּ לְשָׁלוֹם, (במטוס: וְתַטִּיסֵנוּ לְשָׁלוֹם) וְתַגִּיעֵנוּ לִמְחוֹז חֶפְצֵנוּ לְחַיִּים וּלְשִׂמְחָה וּלְשָׁלוֹם [וּתְמַלְטֵנוּ מֵרוּחַ סוֹעָה וָסַעַר]. (אם דעתו לחזור מיד אומר וְתַחְזִירֵנוּ לְשָׁלוֹם) וְתַצִּילֵנוּ מִכַּף כָּל אוֹיֵב וְאוֹרֵב וְלִסְטִים וְחַיּוֹת רָעוֹת בַּדֶּרֶךְ, וּמִכָּל מִינֵי פֻּרְעָנֻיּוֹת הַמִּתְרַגְּשׁוֹת לָבוֹא לָעוֹלָם, וְתִשְלַח בְּרָכָה בְּכל מַעֲשֵה יָדֵינוּ, וְתִתְּנֵנוּ לְחֵן וּלְחֶסֶד וּלְרַחֲמִים בְּעֵינֶיךָ וּבְעֵינֵי כׇל רֹאֵינוּ (וְתִשְמַע קוֹל תַּחֲנוּנֵינוּ), כִּי אל שׁוֹמֵעַ תְּפִלָּה וְתַחֲנוּן אַתָּה. בָּרוּךְ אַתָּה יי שׁוֹמֵעַ תְּפִלָּה:
